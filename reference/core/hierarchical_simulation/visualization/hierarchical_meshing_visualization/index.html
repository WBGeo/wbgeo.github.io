    Hierarchical meshing visualization

    Interactive visualization for hierarchical WBGeo meshes.

    Each hierarchical mesh is displayed in its own PyVista subplot. Every mesh block is added as an independent actor so that it can be switched ON/OFF interactively, similar to ParaView's block list.

    plot_hierarchical_meshes_3d(mesh1, mesh2, geomodel_result, style='surface', show_plotter=True, link_views=True, volume_opacity=1.0, show_block_controls=True, omit_last_blocks=6, enable_inspection=True)

    Plot two hierarchical meshes with interactive block visibility.

    Parameters

    mesh1 Bigger hierarchical mesh.

    mesh2 Smaller hierarchical mesh.

    geomodel_result WBGeo structural model result.

    style Kept for compatibility with the existing WBGeo visualization API.

    show_plotter Immediately open the PyVista window.

    link_views Link the cameras of the two hierarchical views.

    volume_opacity Opacity of 3-D blocks. Set to 1.0 for opaque volumes. Values such as 0.2 make internal surfaces and lines easier to see.

    show_block_controls Add individual block ON/OFF controls.

    omit_last_blocks Number of trailing WBGeo auxiliary blocks to ignore completely. The hierarchical viewer defaults to 6.

    enable_inspection Add ParaView-like interactive inspection. Point mode is enabled by default. Left-click a point to display node ID, XYZ coordinates, block name and point arrays. Press P for point mode and C for cell mode in the currently active subplot. Cell mode reports cell ID, cell type, center, point IDs and cell arrays.

    def plot_hierarchical_meshes_3d(
    mesh1: MeshResults,
    mesh2: MeshResults,
    geomodel_result: StructuralModelResults,
    style: str = "surface",
    show_plotter: bool = True,
    link_views: bool = True,
    volume_opacity: float = 1.0,
    show_block_controls: bool = True,
    omit_last_blocks: int = 6,
    enable_inspection: bool = True,
) -> pv.Plotter:
    """
    Plot two hierarchical meshes with interactive block visibility.

    Parameters
    ----------
    mesh1
        Bigger hierarchical mesh.

    mesh2
        Smaller hierarchical mesh.

    geomodel_result
        WBGeo structural model result.

    style
        Kept for compatibility with the existing WBGeo visualization API.

    show_plotter
        Immediately open the PyVista window.

    link_views
        Link the cameras of the two hierarchical views.

    volume_opacity
        Opacity of 3-D blocks. Set to 1.0 for opaque volumes.
        Values such as 0.2 make internal surfaces and lines easier to see.

    show_block_controls
        Add individual block ON/OFF controls.

    omit_last_blocks
        Number of trailing WBGeo auxiliary blocks to ignore completely.
        The hierarchical viewer defaults to 6.

    enable_inspection
        Add ParaView-like interactive inspection. Point mode is enabled by
        default. Left-click a point to display node ID, XYZ coordinates, block
        name and point arrays. Press ``P`` for point mode and ``C`` for cell
        mode in the currently active subplot. Cell mode reports cell ID, cell
        type, center, point IDs and cell arrays.
    """

    plotter = pv.Plotter(
        shape=(1, 2),
        window_size=(1500, 850),
    )

    # =================================================================
    # Mesh 1
    # =================================================================

    plotter.subplot(0, 0)

    actors1 = _add_blocks(
        plotter,
        mesh1,
        subplot=(0, 0),
        volume_opacity=volume_opacity,
        omit_last_blocks=omit_last_blocks,
    )

    plotter.add_text(
        "Hierarchical mesh 1 (bigger)",
        position="upper_edge",
        font_size=14,
    )

    plotter.show_axes()

    if show_block_controls:

        _add_block_panel(
            plotter,
            actors1,
            subplot=(0, 0),
            start_x=10,
            start_y=700,
        )

    # =================================================================
    # Mesh 2
    # =================================================================

    plotter.subplot(0, 1)

    actors2 = _add_blocks(
        plotter,
        mesh2,
        subplot=(0, 1),
        volume_opacity=volume_opacity,
        omit_last_blocks=omit_last_blocks,
    )

    plotter.add_text(
        "Hierarchical mesh 2 (smaller)",
        position="upper_edge",
        font_size=14,
    )

    plotter.show_axes()

    if show_block_controls:

        _add_block_panel(
            plotter,
            actors2,
            subplot=(0, 1),
            start_x=10,
            start_y=700,
        )

    # =================================================================
    # ParaView-like inspection
    # =================================================================

    if enable_inspection:
        # Point picking is the default inspection mode.  P/C switch the
        # currently active renderer between point and cell inspection.
        _enable_hierarchical_point_mode(
            plotter,
            mesh1,
            mesh2,
            omit_last_blocks,
        )

        def point_mode_active_subplot():
            index = getattr(plotter, "_active_renderer_index", 0)
            if index == 1:
                _enable_point_mode(plotter, mesh2, (0, 1), "Mesh 2", omit_last_blocks)
            else:
                _enable_point_mode(plotter, mesh1, (0, 0), "Mesh 1", omit_last_blocks)

        def cell_mode_active_subplot():
            index = getattr(plotter, "_active_renderer_index", 0)
            if index == 1:
                _enable_cell_mode(plotter, mesh2, (0, 1), "Mesh 2", omit_last_blocks)
            else:
                _enable_cell_mode(plotter, mesh1, (0, 0), "Mesh 1", omit_last_blocks)

        plotter.add_key_event("p", point_mode_active_subplot)
        plotter.add_key_event("c", cell_mode_active_subplot)

    # =================================================================
    # Camera
    # =================================================================

    if link_views:
        plotter.link_views()

    plotter.subplot(0, 0)
    plotter.view_isometric()

    plotter.subplot(0, 1)
    plotter.view_isometric()

    # Keep references available for debugging / external manipulation.
    plotter._hierarchical_actors_mesh1 = actors1
    plotter._hierarchical_actors_mesh2 = actors2

    if show_plotter:
        plotter.show()

    return plotter
